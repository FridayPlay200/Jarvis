_net_lock = threading.Lock()


async def _synthesize(text, path, rate, pitch):
    await edge_tts.Communicate(text, VOICE, rate=rate, pitch=pitch).save(path)


def synth(text, quick=False):
    os.makedirs(VOICE_CACHE, exist_ok=True)
    key = hashlib.md5(f"{text}|{VOICE}|{VOICE_RATE}|{VOICE_PITCH}".encode()).hexdigest()
    mp3 = os.path.join(VOICE_CACHE, key + ".mp3")
    if os.path.exists(mp3):
        return mp3

    attempts = [(VOICE_RATE, VOICE_PITCH), (VOICE_RATE, "+0Hz"), ("+0%", "+0Hz")]
    if quick:
        attempts = attempts[:1]
    tmp = f"{mp3}.{threading.get_ident()}.part"
    error = None
    with _net_lock:
        for rate, pitch in attempts:
            try:
                asyncio.run(_synthesize(text, tmp, rate, pitch))
                if os.path.getsize(tmp) > 0:
                    os.replace(tmp, mp3)
                    return mp3
            except Exception as e:
                error = e
                time.sleep(0.15)
    raise RuntimeError(f"не удалось озвучить: {error}")


def load_sound(path):
    sound = pygame.mixer.Sound(path)
    if not (VOICE_FX and np is not None):
        return sound
    try:
        freq, fmt, channels = pygame.mixer.get_init()
        if fmt != -16:
            return sound
        x = np.frombuffer(sound.get_raw(), dtype=np.int16).astype(np.float32).reshape(-1, channels)
        n = x.shape[0]
        spec = np.fft.rfft(x, axis=0)
        hz = np.fft.rfftfreq(n, 1.0 / freq)
        spec[(hz < 110) | (hz > 8500)] = 0
        y = np.fft.irfft(spec, n=n, axis=0)
        y = np.vstack([y, np.zeros((int(freq * 0.12), channels), dtype=np.float32)])
        dry = y.copy()
        for ms, gain in ((40, 0.18), (95, 0.08)):
            d = int(freq * ms / 1000)
            if d < len(y):
                y[d:] += gain * dry[:-d]
        peak = float(np.abs(y).max()) or 1.0
        y = np.clip(y * (0.9 * 32767 / peak), -32768, 32767).astype(np.int16)
        return pygame.mixer.Sound(buffer=y.tobytes())
    except Exception:
        return sound


def split_sentences(text):
    text = re.sub(r"[*_`#>]", "", text).strip()
    return [s.strip() for s in re.split(r"(?<=[.!?…])\s+", text) if re.search(r"\w", s)]


def _play_synced(text, path):
    if path is None:
        print(f"JARVIS: {text}")
        return
    try:
        sound = load_sound(path)
        length = sound.get_length()
        channel = sound.play()
        busy = channel.get_busy if channel else (lambda: False)
    except pygame.error:
        pygame.mixer.music.load(path)
        pygame.mixer.music.play()
        length = len(text) * 0.06
        busy = pygame.mixer.music.get_busy

    sys.stdout.write("JARVIS: ")
    sys.stdout.flush()
    if TYPE_SYNC and length > 0:
        delay = length * 0.92 / max(len(text), 1)
        start = time.time()
        for i, ch in enumerate(text):
            sys.stdout.write(ch)
            sys.stdout.flush()
            pause = start + (i + 1) * delay - time.time()
            if pause > 0:
                time.sleep(pause)
    else:
        sys.stdout.write(text)
    sys.stdout.write("\n")
    sys.stdout.flush()
    while busy():
        time.sleep(0.02)


class Voice:
    def __init__(self):
        self.synth_q = queue.Queue()
        self.play_q = queue.Queue()
        self.pending = 0
        self.last_activity = time.time()
        self.recent = collections.deque(maxlen=5)
        self.cv = threading.Condition()
        threading.Thread(target=self._synth_loop, daemon=True).start()
        threading.Thread(target=self._play_loop, daemon=True).start()

    def idle(self):
        return self.pending == 0 and time.time() - self.last_activity > 2.5

    def say(self, text):
        self.last_activity = time.time()
        for sentence in split_sentences(text):
            self.recent.append(re.sub(r"[^\w\s]", "", sentence.lower()))
            with self.cv:
                self.pending += 1
            self.synth_q.put(sentence)

    def wait(self):
        with self.cv:
            self.cv.wait_for(lambda: self.pending == 0)

    def _synth_loop(self):
        while True:
            text = self.synth_q.get()
            try:
                path = synth(text)
            except Exception as e:
                print(f"[голос недоступен: {e}]")
                path = None
            self.play_q.put((text, path))

    def _play_loop(self):
        while True:
            text, path = self.play_q.get()
            try:
                _play_synced(text, path)
            except Exception as e:
                print(f"[ошибка воспроизведения: {e}]")
            finally:
                self.last_activity = time.time()
                with self.cv:
                    self.pending -= 1
                    self.cv.notify_all()

def init_vosk():
    global VOSK_MODEL
    try:
        from vosk import Model, SetLogLevel
    except ImportError:
        print("Пакет vosk не установлен (pip install vosk).")
        return False
    try:
        SetLogLevel(-1)
        base = os.path.dirname(os.path.abspath(__file__))
        path = os.path.join(base, VOSK_MODEL_NAME)
        if not os.path.isdir(path):
            print("Скачиваю модель распознавания речи (~45 МБ), это один раз...")
            zip_path = path + ".zip"
            urllib.request.urlretrieve(VOSK_URL, zip_path)
            with zipfile.ZipFile(zip_path) as z:
                z.extractall(base)
            os.remove(zip_path)
        VOSK_MODEL = Model(path)
        return True
    except Exception as e:
        print(f"[Vosk не запустился: {e}]")
        return False


def listen_vosk(timeout, phrase_limit):
    from vosk import KaldiRecognizer
    rec = KaldiRecognizer(VOSK_MODEL, 16000)
    time.sleep(0.2)
    start, speech_start = time.time(), None
    with sr.Microphone(sample_rate=16000, chunk_size=2000) as source:
        while True:
            data = source.stream.read(2000)
            if rec.AcceptWaveform(data):
                text = json.loads(rec.Result()).get("text", "")
                if text:
                    return text
                speech_start = None
            elif speech_start is None and json.loads(rec.PartialResult()).get("partial"):
                speech_start = time.time()

            now = time.time()
            if speech_start is None:
                if timeout and now - start > timeout:
                    return ""
                if not timeout and now - start > 60:
                    rec, start = KaldiRecognizer(VOSK_MODEL, 16000), now
            elif now - speech_start > phrase_limit:
                return json.loads(rec.FinalResult()).get("text", "")


def listen_google(timeout, phrase_limit):
    try:
        with sr.Microphone() as source:
            audio = recognizer.listen(source, timeout=timeout, phrase_time_limit=phrase_limit)
    except sr.WaitTimeoutError:
        return ""
    try:
        return recognizer.recognize_google(audio, language=LANGUAGE)
    except sr.UnknownValueError:
        return ""
    except sr.RequestError as e:
        print(f"[распознавание речи недоступно: {e}]")
        return ""


def listen(timeout=None, phrase_limit=15):
    try:
        text = (listen_vosk if VOSK_MODEL is not None else listen_google)(timeout, phrase_limit)

        time.sleep(1)
        return ""
    text = text.lower().replace("ё", "е").strip()
    if text:
        print(f"Вы: {text}")
    return text


def is_echo(text):
    words = text.split()
    if len(words) < 2:
        return False
    return any(difflib.SequenceMatcher(None, text, r).ratio() > 0.7 for r in voice.recent)


def split_wake(text):
    words = text.split()
    for i, w in enumerate(words):
        w = w.strip(",.!?:;")
        if w in WAKE_WORDS or difflib.get_close_matches(w, WAKE_WORDS, n=1, cutoff=0.75):
            return True, " ".join(words[:i] + words[i + 1:]).strip(" ,.")
    return False, text


def confirm(question):
    voice.say(question)
    voice.wait()
    answer = listen(timeout=6, phrase_limit=4)
    return any(w in answer.split() for w in ("да", "подтверждаю", "давай", "выполняй", "yes"))

def _our_pids():
    pids = {os.getpid()}
    try:
        pids |= {p.pid for p in psutil.Process().parents()}
    except Exception:
        pass
    return pids


def close_active_window():
    if OS == "Darwin":
        pyautogui.hotkey("command", "w")
        return None
    if OS != "Windows":
        pyautogui.hotkey("alt", "f4")
        return None

    import ctypes
    from ctypes import wintypes

    user32 = ctypes.windll.user32
    dwmapi = ctypes.windll.dwmapi
    EnumProc = ctypes.WINFUNCTYPE(ctypes.c_bool, wintypes.HWND, wintypes.LPARAM)

    user32.GetForegroundWindow.restype = wintypes.HWND
    user32.GetWindowThreadProcessId.argtypes = [wintypes.HWND, ctypes.POINTER(wintypes.DWORD)]
    user32.GetWindowTextLengthW.argtypes = [wintypes.HWND]
    user32.GetWindowTextW.argtypes = [wintypes.HWND, wintypes.LPWSTR, ctypes.c_int]
    user32.IsWindowVisible.argtypes = [wintypes.HWND]
    user32.IsIconic.argtypes = [wintypes.HWND]
    user32.GetWindowLongW.argtypes = [wintypes.HWND, ctypes.c_int]
    user32.PostMessageW.argtypes = [wintypes.HWND, wintypes.UINT, wintypes.WPARAM, wintypes.LPARAM]
    user32.EnumWindows.argtypes = [EnumProc, wintypes.LPARAM]

    WM_CLOSE, GWL_EXSTYLE, WS_EX_TOOLWINDOW, DWMWA_CLOAKED = 0x0010, -20, 0x00000080, 14
    IGNORED_TITLES = {"Program Manager"}
    ours = _our_pids()

    def pid_of(hwnd):
        pid = wintypes.DWORD()
        user32.GetWindowThreadProcessId(hwnd, ctypes.byref(pid))
        return pid.value

    def title_of(hwnd):
        n = user32.GetWindowTextLengthW(hwnd)
        buf = ctypes.create_unicode_buffer(n + 1)
        user32.GetWindowTextW(hwnd, buf, n + 1)
        return buf.value

    def is_cloaked(hwnd):
        flag = ctypes.c_int(0)
        dwmapi.DwmGetWindowAttribute(hwnd, DWMWA_CLOAKED, ctypes.byref(flag), ctypes.sizeof(flag))
        return flag.value != 0

    def is_real_window(hwnd):
        title = title_of(hwnd)
        return bool(
            title and title not in IGNORED_TITLES
            and user32.IsWindowVisible(hwnd) and not user32.IsIconic(hwnd)
            and not is_cloaked(hwnd)
            and not (user32.GetWindowLongW(hwnd, GWL_EXSTYLE) & WS_EX_TOOLWINDOW)
        )

    target = user32.GetForegroundWindow()
    if not target or pid_of(target) in ours or not is_real_window(target):
        found = []

        def collect(hwnd, _):
            if pid_of(hwnd) not in ours and is_real_window(hwnd):
                found.append(hwnd)
            return True

        user32.EnumWindows(EnumProc(collect), 0)
        target = found[0] if found else None

    if not target:
        return "Нет окна для закрытия, сэр."
    user32.PostMessageW(target, WM_CLOSE, 0, 0)
    return None


def run_action(action):
    t, a = action.get("type"), action.get("args", {}) or {}

    if t == "open_app":
        cmd = APPS.get(OS, {}).get(str(a.get("name", "")).lower())
        if not cmd:
            return "Не знаю такого приложения, сэр."
        subprocess.Popen(cmd, shell=True)

    elif t == "open_url":
        webbrowser.open(a["url"])

    elif t == "search_web":
        webbrowser.open("https://www.google.com/search?q=" + quote(a["query"]))

    elif t == "type_text":
        text = a["text"]
        pyautogui.write(text, interval=0.02) if text.isascii() else _paste(text)

    elif t == "press_keys":
        pyautogui.hotkey(*a["keys"])

    elif t == "close_window":
        return close_active_window()

    elif t == "media":
        cmd = a.get("command")
        key = MEDIA_KEYS.get(cmd)
        if key:
            for _ in range(5 if cmd in ("volup", "voldown") else 1):
                pyautogui.press(key)

    elif t == "screenshot":
        path = os.path.join(os.path.expanduser("~"), f"screenshot_{datetime.now():%Y%m%d_%H%M%S}.png")
        pyautogui.screenshot(path)

    elif t == "system_info":
        info = (f"Процессор загружен на {psutil.cpu_percent(interval=0.5):.0f} процентов, "
                f"память на {psutil.virtual_memory().percent:.0f}.")
        battery = psutil.sensors_battery()
        if battery:
            info += f" Заряд батареи {battery.percent:.0f} процентов."
        return info

    elif t == "lock_screen":
        cmds = {"Windows": "rundll32.exe user32.dll,LockWorkStation",
                "Darwin": "pmset displaysleepnow", "Linux": "loginctl lock-session"}
        subprocess.Popen(cmds[OS], shell=True)

    elif t == "shell":
        cmd = a["command"]
        if not confirm(f"Выполнить команду: {cmd}? Подтвердите."):
            return "Отменено, сэр."
        result = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=60)
        print(result.stdout or result.stderr)
        return "Команда выполнена." if result.returncode == 0 else "Команда завершилась с ошибкой."

    elif t == "shutdown":
        if not confirm("Вы уверены, что хотите выключить компьютер?"):
            return "Отменено, сэр."
        cmds = {"Windows": "shutdown /s /t 5", "Darwin": "sudo shutdown -h now", "Linux": "shutdown now"}
        subprocess.Popen(cmds[OS], shell=True)

    return None

def _match_segment(s):
    s = s.strip(" ,.!?")

    if re.search(r"сколько времени|который час|скажи время|текущее время", s):
        return None, f"Сейчас {datetime.now():%H:%M}, сэр."
    if re.search(r"какое (сегодня )?число|какая (сегодня )?дата|какой (сегодня )?день", s):
        n = datetime.now()
        return None, f"Сегодня {WEEKDAYS[n.weekday()]}, {n.day} {MONTHS[n.month - 1]}, сэр."

    if re.search(r"закр\w*\s+(это\s+|текущее\s+|активное\s+|открытое\s+)?(окно|приложение|программу)"
                 r"|^закр\w*(\s+это)?$", s):
        return {"type": "close_window", "args": {}}, "Закрываю."

    m = re.search(r"(?:найди|поищи|включи)\s+(.+?)\s+(?:в|на)\s+(?:ютубе|ютюбе|youtube)", s)
    if m:
        url = "https://www.youtube.com/results?search_query=" + quote(m.group(1))
        return {"type": "open_url", "args": {"url": url}}, "Ищу, сэр."

    m = re.search(r"(?:найди|поищи|загугли|погугли|поиск)\s+(.+)", s)
    if m:
        return {"type": "search_web", "args": {"query": m.group(1)}}, "Ищу, сэр."

    m = re.search(r"(?:напиши|набери|введи|напечатай)\s+(.+)", s)
    if m:
        return {"type": "type_text", "args": {"text": m.group(1)}}, "Печатаю."

    if re.search(r"скриншот|снимок экрана|сфотографируй экран", s):
        return {"type": "screenshot", "args": {}}, "Делаю снимок."
    if re.search(r"заряд|загрузк|состояние системы|как (там )?система", s):
        return {"type": "system_info", "args": {}}, ""
    if re.search(r"заблокируй|блокировка экрана", s):
        return {"type": "lock_screen", "args": {}}, "Блокирую."
    if re.search(r"выключи (компьютер|пк|систему)|заверши работу", s):
        return {"type": "shutdown", "args": {}}, ""

    keys = [
        (r"сверни (все )?окна|покажи рабочий стол", ["win", "d"], "Выполняю."),
        (r"(открой )?нов(ую|ая) вкладк", ["ctrl", "t"], "Открываю."),
        (r"закрой вкладку", ["ctrl", "w"], "Закрываю."),
        (r"обнови страницу|перезагрузи страницу", ["f5"], "Обновляю."),
        (r"^скопируй", ["ctrl", "c"], "Копирую."),
        (r"^вставь", ["ctrl", "v"], "Вставляю."),
        (r"выдели все|выбери все", ["ctrl", "a"], "Выделяю."),
        (r"^сохрани", ["ctrl", "s"], "Сохраняю."),
        (r"^отмени|верни назад", ["ctrl", "z"], "Отменяю."),
        (r"полный экран", ["f11"], "Выполняю."),
        (r"переключи окно|следующее окно", ["alt", "tab"], "Переключаю."),
        (r"прокрути вниз|листай вниз", ["pagedown"], "Выполняю."),
        (r"прокрути вверх|листай вверх", ["pageup"], "Выполняю."),
    ]
    for pattern, combo, say in keys:
        if re.search(pattern, s):
            return {"type": "press_keys", "args": {"keys": combo}}, say

    media = [
        (r"без звука|выключи звук|отключи звук", "mute", "Звук отключён."),
        (r"громче|прибавь звук|увеличь громкость", "volup", "Громче."),
        (r"тише|убавь звук|уменьши громкость", "voldown", "Тише."),
        (r"пауза|поставь на паузу|продолжи|воспроизведи|играй", "playpause", "Выполняю."),
        (r"следующ", "next", "Следующий трек."),
        (r"предыдущ|прошлый трек", "prev", "Предыдущий трек."),
    ]
    for pattern, cmd, say in media:
        if re.search(pattern, s):
            return {"type": "media", "args": {"command": cmd}}, say

    m = re.search(r"(?:открой|запусти|включи|зайди на|зайди в)\s+(.+)", s)
    if m:
        target = m.group(1)
        for site, url in SITE_ALIASES.items():
            if site in target:
                return {"type": "open_url", "args": {"url": url}}, "Открываю."
        for aliases, app in APP_ALIASES:
            if any(al in target for al in aliases):
                return {"type": "open_app", "args": {"name": app}}, "Открываю."
        return {"type": "search_web", "args": {"query": target}}, "Ищу в интернете, сэр."

    if re.search(r"курс|погод|новости|рецепт|что такое|кто такой", s):
        return {"type": "search_web", "args": {"query": s}}, "Ищу в интернете, сэр."

    if re.search(r"^(привет|здравствуй|доброе утро|добрый день|добрый вечер)", s):
        return None, "Приветствую, сэр. Чем могу быть полезен?"
    if re.search(r"как (у тебя |твои )?дела|как ты|как жизнь|как самочувствие", s):
        return None, "Все системы в норме, сэр."
    if re.search(r"спасибо|благодарю", s):
        return None, "Всегда к вашим услугам, сэр."
    return False


def rule_based(text):
    parts = re.split(r"\s+(?:и потом|и затем|потом|затем|после этого|и)\s+", text)
    results = [_match_segment(p) for p in parts] if len(parts) > 1 else []

    if results and all(r is not False for r in results):
        actions = [r[0] for r in results if r[0]]
        say = "Выполняю, сэр." if len(results) > 1 else results[0][1]
        return {"say": say, "actions": actions}

    single = _match_segment(text)
    if single is not False:
        return {"say": single[1], "actions": [single[0]] if single[0] else []}
    return None


# нейросеть (Ollama)

def _visible(buf):
    if buf.lstrip().startswith("{"):
        return ""
    up = buf.upper()
    i = up.find("ACTION")
    if i != -1:
        return buf[:i]
    for k in range(5, 0, -1):
        if up.endswith("ACTION"[:k]):
            return buf[:-k]
    return buf


def _emit_sentences(visible, start, say):
    pos = start
    while True:
        m = re.search(r"[.!?…]+\s", visible[pos:])
        if not m:
            return pos
        sentence = visible[pos:pos + m.end()].strip()
        pos += m.end()
        if sentence:
            say(sentence)


def ask_llm(user_text):
    global _llm_warned
    history.append({"role": "user", "content": user_text})
    payload = {
        "model": OLLAMA_MODEL,
        "messages": [{"role": "system", "content": SYSTEM_PROMPT}] + history[-6:],
        "stream": True,
        "keep_alive": "30m",
        "options": {"temperature": 0.3, "num_predict": 100, "num_ctx": 1024},
    }
    req = urllib.request.Request(
        OLLAMA_URL, data=json.dumps(payload).encode(), headers={"Content-Type": "application/json"}
    )
    buf, spoken = "", 0
    try:
        with urllib.request.urlopen(req, timeout=60) as r:
            for line in r:
                if not line.strip():
                    continue
                chunk = json.loads(line)
                buf += chunk.get("message", {}).get("content", "")
                spoken = _emit_sentences(_visible(buf), spoken, voice.say)
                if chunk.get("done"):
                    break
    except urllib.error.HTTPError as e:
        history.pop()
        if e.code == 404:
            print(f"Модель не скачана. Выполните в терминале:  ollama pull {OLLAMA_MODEL}")
            voice.say("Модель нейросети не скачана, сэр. Команда указана в консоли.")
        else:
            voice.say("Нейросеть вернула ошибку, сэр.")
        return []
    except (OSError, json.JSONDecodeError):
        if not buf:
            history.pop()
            print("Ollama не запущена. Установите её с https://ollama.com/download и перезапустите Джарвиса.")
            voice.say("Нейросеть не запущена, сэр. Простые команды я выполню и без неё."
                      if not _llm_warned else "Этого я без нейросети не умею, сэр.")
            _llm_warned = True
            return []

    actions, text = [], buf
    if buf.lstrip().startswith("{"):
        try:
            data = json.loads(buf)
            text = str(data.get("say", ""))
            act = data.get("action")
            actions = [act] if isinstance(act, dict) else []
            spoken = 0
        except ValueError:
            text = re.sub(r"[{}\"]", "", buf)
    else:
        m = re.search(r"(?is)ACTION:\s*(\{.*\})", buf)
        if m:
            try:
                act = json.loads(m.group(1))
                actions = [act] if isinstance(act, dict) and act.get("type") else []
            except ValueError:
                pass
        text = re.split(r"(?i)ACTION", buf)[0]

    tail = text[spoken:].strip()
    if tail:
        voice.say(tail)
    elif not text.strip() and not actions:
        voice.say("Не понял команду, сэр.")
    history.append({"role": "assistant", "content": buf.strip()})
    return actions


def start_ollama():
    def alive():
        try:
            urllib.request.urlopen(OLLAMA_BASE, timeout=1.5)
            return True
        except Exception:
            return False

    if not alive():
        exe = shutil.which("ollama")
        if not exe:
            print("Ollama не найдена - работают только команды по правилам. "
                  "Скачать: https://ollama.com/download")
            return
        flags = subprocess.CREATE_NO_WINDOW if OS == "Windows" else 0
        subprocess.Popen([exe, "serve"], stdout=subprocess.DEVNULL,
                         stderr=subprocess.DEVNULL, creationflags=flags)
        for _ in range(30):
            time.sleep(0.5)
            if alive():
                break
        else:
            return

    try:
        req = urllib.request.Request(
            OLLAMA_BASE + "/api/generate",
            data=json.dumps({"model": OLLAMA_MODEL, "prompt": "", "keep_alive": "30m"}).encode(),
            headers={"Content-Type": "application/json"},
        )
        urllib.request.urlopen(req, timeout=180)
        print(f"Нейросеть {OLLAMA_MODEL} готова.")
    except urllib.error.HTTPError as e:
        if e.code == 404:
            print(f"Модель не скачана. Выполните в терминале:  ollama pull {OLLAMA_MODEL}")
    except Exception:
        pass


# главный цикл

def handle(command, strict=False):
    decision = rule_based(command)
    if decision is not None:
        actions = decision["actions"]
        if decision["say"]:
            voice.say(decision["say"])
            if any(a.get("type") in ("shell", "shutdown") for a in actions):
                voice.wait()
    elif strict and (not USE_LLM or len(command.split()) < 2):
        return
    elif USE_LLM:
        actions = ask_llm(command)
    else:
        actions, _ = [], voice.say("Не понял команду, сэр.")

    for action in actions:
        extra = run_action(action)
        if extra:
            voice.say(extra)
    voice.wait()


EXIT_WORDS = {"отбой", "выход", "пока"}
SLEEP_RE = re.compile(r"режим ожидания|не слушай|помолчи|усни")


def main():
    global STT_ENGINE
    if STT_ENGINE == "vosk" and not init_vosk():
        print("Использую распознавание Google (медленнее).")
        STT_ENGINE = "google"
    if VOSK_MODEL is None:
        with sr.Microphone() as source:
            recognizer.adjust_for_ambient_noise(source, duration=1)
        recognizer.energy_threshold = min(max(recognizer.energy_threshold, 200), 2000)

    threading.Thread(target=prewarm_voice, daemon=True).start()
    if USE_LLM:
        threading.Thread(target=start_ollama, daemon=True).start()
    voice.say("Системы запущены. Слушаю вас, сэр.")
    voice.wait()

    sleeping = False
    while True:
        try:
            heard = listen(phrase_limit=10)
            if not heard or is_echo(heard):
                continue
            woke, command = split_wake(heard)

            if sleeping:
                if not woke:
                    continue
                sleeping = False
            elif not woke and WAKE_WORD_REQUIRED:
                continue

            if woke and not command:
                voice.say("Да, сэр?")
                voice.wait()
                command = listen(timeout=8)
                if not command or is_echo(command):
                    continue

            if EXIT_WORDS & set(command.split()) or "до свидания" in command:
                voice.say("Хорошего дня, сэр.")
                voice.wait()
                break
            if SLEEP_RE.search(command):
                sleeping = True
                voice.say("Перехожу в режим ожидания. Назовите моё имя, когда понадоблюсь, сэр.")
                voice.wait()
                continue

            handle(command, strict=not woke)
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[ошибка: {e}]")
            voice.say("Произошла ошибка, сэр.")
            voice.wait()


if __name__ == "__main__":
    main()
